-- ----------------------------------------------------------------------------
-- 1. O envelope tolerante
-- ----------------------------------------------------------------------------
-- Assinatura e RETURNS TABLE identicos aos de fn_blocos_previstos_dia (vigente em
-- 20260908140000). Nao replica regra nenhuma: so chama e engole a recusa de escopo.
--
-- STABLE e SECURITY DEFINER como a original. SECURITY DEFINER aqui NAO burla o
-- guard: auth.uid() le o JWT do request, nao o usuario do banco, entao a checagem
-- de dentro de fn_blocos_previstos_dia continua valendo por inteiro - o que muda e
-- so o que fazemos com a recusa.
CREATE OR REPLACE FUNCTION public.fn_blocos_previstos_dia_vizinho(
    p_servidor_id uuid,
    p_data        date
)
RETURNS TABLE (
    bloco_ordem               integer,
    escala_diaria_ids         uuid[],
    categoria                 text,
    inicio_previsto           timestamptz,
    fim_previsto              timestamptz,
    intervalo_inicio_previsto timestamptz,
    intervalo_fim_previsto    timestamptz,
    permite_intervalo         boolean,
    turnos_inicio             timestamptz[],
    turnos_fim                timestamptz[]
)
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = public
AS $fnviz$
BEGIN
    RETURN QUERY
        SELECT b.bloco_ordem, b.escala_diaria_ids, b.categoria,
               b.inicio_previsto, b.fim_previsto,
               b.intervalo_inicio_previsto, b.intervalo_fim_previsto,
               b.permite_intervalo, b.turnos_inicio, b.turnos_fim
          FROM public.fn_blocos_previstos_dia(p_servidor_id, p_data) b;
EXCEPTION
    -- Dia vizinho fora do escopo de quem chama: nenhum bloco de ontem. Nunca um erro.
    WHEN insufficient_privilege THEN
        RETURN;
END;
$fnviz$;

COMMENT ON FUNCTION public.fn_blocos_previstos_dia_vizinho(uuid, date) IS
    'fn_blocos_previstos_dia para o dia VIZINHO: devolve vazio quando o guard de escopo '
    'recusa aquele dia (servidor sem escala no mes anterior/seguinte dentro do escopo de '
    'quem chama), em vez de derrubar a alocacao do dia consultado. 09/09/2026.';

-- Objeto NOVO nasce com EXECUTE para PUBLIC (armadilhas 24/41): o REVOKE tem que ser
-- escrito aqui, agora, que e quando somos comprovadamente o dono.
REVOKE ALL ON FUNCTION public.fn_blocos_previstos_dia_vizinho(uuid, date) FROM PUBLIC;
REVOKE ALL ON FUNCTION public.fn_blocos_previstos_dia_vizinho(uuid, date) FROM anon;
GRANT EXECUTE ON FUNCTION public.fn_blocos_previstos_dia_vizinho(uuid, date)
    TO authenticated, service_role;
